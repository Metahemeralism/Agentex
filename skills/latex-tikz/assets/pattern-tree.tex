    level distance=15mm,

    leaf/.style={term, minimum width=10mm, fill=gray!30}]

    child {node[split] {$x_2 \le 1.2$}

      child {node[leaf] {B} edge from parent node[right, font=\footnotesize] {no}}
      edge from parent node[left, font=\footnotesize] {yes}}
    child {node[leaf] {C} edge from parent node[right, font=\footnotesize] {no}};

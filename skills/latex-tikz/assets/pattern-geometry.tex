% Geometric figure with exact construction: calc projection, angle and right-angle marks.
\documentclass[tikz, border=6pt]{standalone}
\usepackage{amsmath, bm}
\usepackage{agentex-tikz}
\usetikzlibrary{angles}
\begin{document}
\begin{tikzpicture}[ax, scale=1.3]
  \coordinate (O) at (0,0);
  \coordinate (A) at (3,0.8);
  \coordinate (B) at (1.4,2);
  \coordinate (P) at ($(O)!(B)!(A)$);  % orthogonal projection of B onto line OA
  \draw[->] (O) -- (A) node[right] {$\bm u$};
  \draw[->] (O) -- (B) node[above] {$\bm v$};
  \draw[aux] (B) -- (P);
  \draw[->, very thick, axblue] (O) -- (P);
  \node[below=1mm, text=axblue] at ($(O)!0.6!(P)$) {$\operatorname{proj}_{\bm u}\bm v$};
  \pic[draw, "$\theta$", angle radius=8mm, angle eccentricity=1.35] {angle = A--O--B};
  \pic[draw, angle radius=2mm] {right angle = O--P--B};
  \fill (O) circle (1pt) node[below left] {$O$};
\end{tikzpicture}
\end{document}

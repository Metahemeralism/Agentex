% Finite-state machine / Markov chain with the automata library.
\documentclass[tikz, border=6pt]{standalone}
\usepackage{amsmath}
\usepackage{agentex-tikz}
\usetikzlibrary{automata}
\begin{document}
\begin{tikzpicture}[ax, node distance=24mm, on grid, auto, shorten >=1pt,
    every state/.style={draw, fill=axfill, minimum size=9mm}]
  \node[state, initial, initial text={}] (q0) {$q_0$};
  \node[state, right=of q0] (q1) {$q_1$};
  \node[state, accepting, right=of q1] (q2) {$q_2$};
  \path[->]
    (q0) edge node {$a$} (q1)
         edge[loop above] node {$b$} ()
    (q1) edge[bend left] node {$b$} (q2)
         edge[loop above] node {$a$} ()
    (q2) edge[bend left] node {$a$} (q1);
\end{tikzpicture}
\end{document}
